% Shared TikZ style of the Reverb paper figures, after the STATOR (LLM-Share) figures: Times text at 8 pt (printed at
% about 7 pt, the size of the tick labels of the graphs), thin lines, one fill colour per kind of object, regular
% weight everywhere except the panel titles.  The PDFs are deterministic (no date, no trailer id) so that a rebuild
% gives the same bytes.
\pdfinfoomitdate=1
\pdftrailerid{}
\usepackage{newtxtext}
\usepackage{tikz}
\newcommand{\figsize}{\fontsize{8}{9.5}\selectfont}
\usetikzlibrary{arrows.meta, positioning, fit, calc, patterns, decorations.pathreplacing}

\definecolor{panel}{HTML}{F3F4F6}    % a group of components
\definecolor{dlc}{HTML}{FFFFFF}      % a downlink slot
\definecolor{spc}{HTML}{D9D9D6}      % a special slot
\definecolor{ulc}{HTML}{7FCBC1}      % an uplink slot
\definecolor{sampc}{HTML}{BFD6F2}    % received samples
\definecolor{gpuc}{HTML}{E4E0F6}     % the GPU and its memory
\definecolor{l1c}{HTML}{D6D5CF}      % Layer-1 processing
\definecolor{aic}{HTML}{F7B955}      % AI inference
\definecolor{actc}{HTML}{FDE9C9}     % an operation of the base station
\definecolor{failc}{HTML}{C5432F}    % a failed CRC and its retransmission
\definecolor{okc}{HTML}{14786F}      % a decoded TB

\tikzset{
  layer/.style={draw=black!60, fill=panel, line width=0.5pt},
  comp/.style={draw=black!85, fill=white, line width=0.4pt, align=center,
               font=\figsize, inner xsep=3pt, inner ysep=1.5pt, minimum height=0.5cm},
  slot/.style={draw=black!65, line width=0.35pt, minimum width=0.42cm, minimum height=0.36cm,
               inner sep=0pt, font=\figsize},
  tag/.style={font=\figsize, inner sep=1pt},
  ttl/.style={font=\bfseries\small},
  note/.style={font=\itshape\figsize, inner sep=1pt},
  arr/.style={-{Latex[length=1.6mm]}, line width=0.8pt},
  thin/.style={-{Latex[length=1.3mm]}, line width=0.5pt},
  num/.style={circle, fill=black, text=white, font=\bfseries\figsize, inner sep=0.6pt, minimum size=0.33cm},
}
% A legend mark for a legend that is one centred line of text:  \node[tag] {\lgdfill{ulc}\,Uplink \quad ...};
\newcommand{\lgdfill}[1]{\tikz[baseline=-0.55ex]{\draw[black!65, line width=0.35pt, fill=#1] (0,-0.1) rectangle (0.22,0.1);}}
