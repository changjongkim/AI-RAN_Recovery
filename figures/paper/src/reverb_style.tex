% Shared TikZ style of the Reverb paper figures (2026-10-29, second version).  The look of the deck figures
% (figures/deck/src/tikz_style2.tex) at the size of the one-column page, in the font of the data panels:
%   - DejaVu Sans everywhere (make_paper_figures.py and the STATOR data panels use it), built with LuaLaTeX
%   - a part of the system is a tinted panel (pan) with a bold title; inside it, white cards (comp) with a navy border,
%     a bold title and a black numbered circle (num) for a step of the procedure
%   - state is drawn as tables with a blue header row (th) and example values (td), or as rows of cells (cl)
%   - stages are joined by fat block arrows (barrow); an action is a yellow box (act), a decision a mint box (dec)
%   - red (dred) only for failed code blocks, retransmissions and the key value of a figure
% One colour per entity in every figure: uplink slot teal, special slot gray, L1 gray-brown, AI blue, received samples
% violet, published large NRx light orange, interference-trained NRx dark orange (the two NRx colours of the data panels).
% Canvas: at most 13.9 cm wide (the text width), and as low as the content allows.  The PDFs carry no date or id.
\pdfvariable suppressoptionalinfo 1023

% A failed TB as its ten code blocks: \cbs{x}{y}{failed}{pitch}{height} (left end x, centre y; failed blocks red).
\newcommand{\cbs}[5]{%
  \foreach \i in {1,...,10} {
    \pgfmathsetmacro{\f}{ifthenelse(\i <= #3, 1, 0)}
    \ifdim \f pt = 1pt \def\fc{dred}\else\def\fc{white}\fi
    \draw[black!70, line width=0.3pt, fill=\fc] ({#1 + #4*(\i-1)}, {#2 - #5/2}) rectangle ({#1 + #4*\i - 0.03}, {#2 + #5/2});
  }}
