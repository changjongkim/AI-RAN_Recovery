% Figure (Section 4.1, b): the recovery deadline of a failed TB on the DDDSU slots.  The TB arrives in an uplink slot;
% its L1 result (CRC fail) is known at t_L1.  Without recovery, the grant follows at t_L1 and the retransmission uses the
% first uplink slot U_s at least the grant lead L later.  With recovery, the large NRx runs for r and its result arrives at
% t_R; the base station may hold the grant until D = U_r - L, where U_r is the latest uplink slot it allows for the
% retransmission, so the retransmission of a TB that recovery does not decode still uses U_r at the latest.  Geometry
% follows the timing of the evaluation (Section 5.1) but no value is printed (Design section rule).

\input{reverb_style}
\newcommand{\tL}{\textit{t}\textsubscript{L1}}
\newcommand{\tR}{\textit{t}\textsubscript{R}}
\newcommand{\Us}{\textit{U}\textsubscript{s}}
\newcommand{\Ur}{\textit{U}\textsubscript{r}}
\begin{document}
\begin{tikzpicture}[x=1cm, y=1cm]
  \foreach \y/\t in {0/Slots, -0.52/Without recovery, -1.0/Recovery decodes, -1.48/Recovery fails} {
    \node[rowh, minimum width=2.3cm, minimum height=0.36cm, font=\tabsize\bfseries] at (1.15, \y) {\t};
  }
  \begin{scope}[xshift=2.72cm, x=0.665cm]
    \def\tl{1.9} \def\tr{8.2} \def\us{7.5} \def\ur{15.0} \def\dd{11.0} \def\uf{12.5}

    \foreach \k in {-1,...,31} {
      \pgfmathsetmacro{\t}{0.5*\k}
      \pgfmathsetmacro{\m}{mod(\k+10, 5)}
      \ifdim \m pt = 0pt \def\c{ulc}\else\ifdim \m pt = 4pt \def\c{spc}\else\def\c{white}\fi\fi
      \draw[black!60, line width=0.3pt, fill=\c] (\t, -0.18) rectangle (\t + 0.5, 0.18);
    }
    \draw[black, line width=1.2pt] (0, -0.18) rectangle (0.5, 0.18);
    \node[btxt, anchor=south] at (0.25, 0.22) {Failed TB};
    % events
    \foreach \t/\l in {\tl/\tL, \tr/\tR, \dd/\vD} {
      \draw[black!70, line width=0.5pt, densely dashed] (\t, -1.7) -- (\t, 0.22);
      \node[btxt, anchor=south] at (\t, 0.22) {\l};
    }
    \draw[dred, line width=1.2pt, densely dashed] (\ur+0.03, -0.15) rectangle (\ur+0.47, 0.15);
    \draw[black!70, line width=0.5pt, densely dashed] (\ur, -0.7) -- (\ur, -0.18);
    \node[btxt, anchor=south] at (\ur + 0.25, 0.22) {\Ur};
    \draw[{Latex[length=1.3mm]}-{Latex[length=1.3mm]}, line width=0.6pt] (\dd, -0.52) -- (\ur, -0.52);
    \node[btxt, fill=white, inner sep=1pt] at ({(\dd+\ur)/2}, -0.52) {Grant lead \vL};
    % without recovery: the grant follows the L1 result
    \draw[arr] (\tl, -0.52) -- (\us, -0.52) node[pos=0.5, txt, above=-1pt] {Grant};
    \draw[dred, line width=1.2pt, fill=ulc] (\us+0.03, -0.68) rectangle (\us + 0.47, -0.36);
    \node[btxt, anchor=west] at (\us + 0.55, -0.52) {\Us};

    \node[cl, fill=nrxc, minimum height=0.36cm, minimum width={0.665*(\tr-\tl)*1cm}, anchor=west, font=\figsize\bfseries] at (\tl, -1.0) {Recovery, \vr};

  \end{scope}
