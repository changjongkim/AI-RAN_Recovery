% Figure (Section 4.1, b): the recovery deadline of a failed TB on the DDDSU slots.  The L1 result (CRC fail) of the TB
% is known at t_L1.  U_r is the latest uplink slot the base station allows for the retransmission, and a grant must
% leave at least the grant lead L before it, so the grant can wait until D = U_r - L.  A recovery that ends by D lets a
% TB that it does not decode still be retransmitted in U_r.  Geometry follows the timing of the evaluation (Section 5.1)
% but no value is printed (Design section rule).
\documentclass[border=1pt]{standalone}
\input{reverb_style}
\newcommand{\tL}{\textit{t}\textsubscript{L1}}
\newcommand{\Ur}{\textit{U}\textsubscript{r}}
\begin{document}
\begin{tikzpicture}[x=1cm, y=1cm]
  \node[exrow, minimum width=2.3cm, minimum height=0.34cm] at (1.15, 0) {Uplink slots};
  \node[rrow, minimum width=2.3cm, minimum height=0.34cm] at (1.15, -0.46) {Recovery};
  \node[exrow, minimum width=2.3cm, minimum height=0.34cm] at (1.15, -0.92) {Grant};
  \begin{scope}[xshift=2.72cm, x=0.665cm]
    \def\tl{1.9} \def\ur{15.0} \def\dd{11.0}
    \foreach \k in {-1,...,31} {
      \pgfmathsetmacro{\t}{0.5*\k}
      \pgfmathsetmacro{\m}{mod(\k+10, 5)}
      \ifdim \m pt = 0pt \def\c{ulc}\else\ifdim \m pt = 4pt \def\c{spc}\else\def\c{white}\fi\fi
      \draw[black!55, line width=0.3pt, fill=\c] (\t, -0.17) rectangle (\t + 0.5, 0.17);
    }
    \draw[black, line width=1.1pt] (0, -0.17) rectangle (0.5, 0.17);
    \node[txt, anchor=south] at (0.25, 0.2) {Failed TB};
    \foreach \t/\l in {\tl/\tL, \dd/\vD} {
      \draw[black!70, line width=0.5pt, densely dashed] (\t, -1.1) -- (\t, 0.2);
      \node[txt, anchor=south] at (\t, 0.2) {\l};
    }
    \draw[dred, line width=1.1pt, fill=ulc] (\ur+0.04, -0.14) rectangle (\ur+0.46, 0.14);
    \node[txt, anchor=south] at (\ur + 0.25, 0.2) {\Ur};
    \draw[black!70, line width=0.5pt, densely dashed] (\ur, -1.1) -- (\ur, -0.17);
    % the recovery may run from the L1 result until D
    \fill[revfill] (\tl, -0.62) rectangle (\dd, -0.3);
    \node[cl, fill=nrxc, minimum height=0.3cm, minimum width={0.665*6.3cm}, anchor=west] at (\tl, -0.46) {Recovery, run time \vr};
    \node[ann, anchor=east] at (\dd - 0.1, -0.46) {Ends by \vD};
    % the latest grant that keeps the retransmission in U_r
    \draw[{Latex[length=1.3mm]}-{Latex[length=1.3mm]}, line width=0.6pt] (\dd, -0.92) -- (\ur, -0.92);
    \node[txt, fill=white, inner sep=1pt] at ({(\dd+\ur)/2}, -0.92) {Grant lead \vL};
    \node[ann, anchor=east] at (\dd - 0.1, -0.92) {Latest grant};
  \end{scope}
  \node[key, anchor=north east] at (13.9, -1.18) {Recovery deadline \vD\,=\,\Ur\,−\,\vL};
  \node[txt, font=\tabsize, anchor=north west] at (0, -1.2) {\lgd{ulc}\,Uplink\enspace \lgd{spc}\,Special\enspace
    \lgd{white}\,Downlink\enspace \lgdrtx\,Latest retransmission slot};
\end{tikzpicture}
\end{document}
