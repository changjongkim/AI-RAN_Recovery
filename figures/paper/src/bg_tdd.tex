% Figure (Section 2.1, a): the two TDD patterns of the 3.5 GHz bands over 5 ms (ten slots of 0.5 ms): DDDSU twice and
% DDDDDDDSUU once.  Both give two of every ten slots to the uplink.  Sits in a minipage of 0.26 of the text width.
\documentclass[border=1pt]{standalone}
\input{reverb_style}
\begin{document}
\begin{tikzpicture}[x=1cm, y=1cm]
  \def\p{0.34}
  \foreach \name/\pat/\y in {DDDSU/{D,D,D,S,U,D,D,D,S,U}/1.45, DDDDDDDSUU/{D,D,D,D,D,D,D,S,U,U}/0.45} {
    \node[btxt, anchor=south west] at (0, \y + 0.2) {\name};
    \foreach \s [count=\i from 0] in \pat {
      \if U\s\def\fc{ulc}\else\if S\s\def\fc{spc}\else\def\fc{white}\fi\fi
      \node[cl, fill=\fc, minimum width=\p cm, minimum height=0.36cm, font=\fontsize{6.2}{7}\selectfont] at ({\p*\i + \p/2}, \y) {\s};
    }
  }
  \draw[{Latex[length=1.3mm]}-{Latex[length=1.3mm]}, line width=0.6pt] (0, 0.0) -- (3.4, 0.0);
  \node[txt, fill=white, inner xsep=2pt] at (1.7, 0.0) {5 ms};
  \node[stxt, anchor=north] at (1.7, -0.22) {\lgd{ulc}\,Uplink\hspace{3pt}\lgd{spc}\,Special\hspace{3pt}\lgd{white}\,Downlink};
\end{tikzpicture}
\end{document}
