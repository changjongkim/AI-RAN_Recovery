% Figure (Section 4.3): admission by the latest start time on one GPU.  Failed TBs 1 and 2 come from uplink slot 0,
% TB 3 from slot 1 and TB 4 from slot 2.  Each TB can start its recovery from its L1 result until its latest start time
% D - r and must finish by its deadline D (11.0 ms after its slot; the run time is drawn as the 4.3 ms of the network
% alone to keep the example short).  In arrival order without rejection, TB 3 starts after its latest start, runs past
% its deadline and holds the GPU, so TB 4 also ends past its deadline: 2 of 4 TBs are recovered in time.  Reverb rejects
% TB 3 at its latest start (its grant goes out at once) and recovers TB 4: 3 of 4 in time.  With the same deadline rule
% in every cell, the deadline order equals the arrival order; the difference is the rejection.  No value is printed.

    \node[ctxrow, minimum width=2.3cm, minimum height=0.24cm] at (1.15, \y) {\t};
  }
  \node[exrow, minimum width=2.3cm, minimum height=0.44cm] at (1.15, -1.2) {Arrival order\\[-1pt]2 of 4 in time};
  \node[rrow, minimum width=2.3cm, minimum height=0.44cm] at (1.15, -1.7) {Reverb\\[-1pt]3 of 4 in time};
  \begin{scope}[xshift=2.45cm, x=0.585cm]
    \def\r{4.3}
    % the window of each TB: from its L1 result it can start until D - r, and it must end by D

      \fill[black] (\a, \y) circle (0.045);
      \draw[dred, line width=0.9pt, densely dashed] (\s, \y-0.1) -- (\s, \y+0.1);
      \draw[dred, line width=1.1pt] (\d, \y-0.1) -- (\d, \y+0.1);
    }
    \newcommand{\job}[3]{% start, lane y, label
      \node[cl, fill=nrxc, minimum width={0.585*\r cm - 0.04cm}, minimum height=0.3cm, anchor=west] at (#1 + 0.035, #2) {#3};}
    % arrival order without rejection
    \node[key, font=\tabsize\itshape, anchor=west] at (10.4, -0.88) {TB 3 starts after its latest start and holds the GPU};

    \fill[pattern=north east lines, pattern color=dred] (13.5, -1.35) rectangle (14.8, -1.05);
    \fill[pattern=north east lines, pattern color=dred] (16.0, -1.35) rectangle (19.1, -1.05);

    \node[txt] at (15.4, -1.2) {4};
    % Reverb: TB 3 rejected at its latest start, TB 4 in time
    \job{1.9}{-1.7}{1} \job{6.2}{-1.7}{2} \job{10.5}{-1.7}{4}
    \node[stxt, text=dgreen, anchor=west] at (14.95, -1.7) {✓ TB 4 ends before its \vD};
